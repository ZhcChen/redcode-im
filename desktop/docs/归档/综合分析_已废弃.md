```typescript
// src/composables/useTheme.ts
export function useTheme() {
  const theme = ref<'light' | 'dark' | 'auto'>('auto')

  const applyTheme = (t: 'light' | 'dark') => {
    document.documentElement.setAttribute('data-theme', t)
  }

  const toggleTheme = () => {
    theme.value = theme.value === 'light' ? 'dark' : 'light'
    applyTheme(theme.value)
  }

  return { theme, toggleTheme, applyTheme }
}
```

```rust
use tauri_plugin_keychain::{Keychain, KeychainProvider};

pub async fn save_token(token: &str) -> Result<(), String> {
    let keychain = Keychain::default();
    keychain.set_password("chatly", "auth_token", token)
        .map_err(|e| e.to_string())
}

pub async fn get_token() -> Result<String, String> {
    let keychain = Keychain::default();
    keychain.get_password("chatly", "auth_token")
        .map_err(|e| e.to_string())
}

pub async fn delete_token() -> Result<(), String> {
    let keychain = Keychain::default();
    keychain.delete_password("chatly", "auth_token")
        .map_err(|e| e.to_string())
}
```

#### 密码前端加密
```typescript
// src/utils/crypto.ts
import { hash } from '@tauri-apps/plugin-crypto'

export async function hashPassword(password: string): Promise<string> {
  const encoder = new TextEncoder()
  const data = encoder.encode(password)
  const hashBuffer = await hash('sha256', data)
  const hashArray = Array.from(new Uint8Array(hashBuffer))
  return hashArray.map(b => b.toString(16).padStart(2, '0')).join('')
}
```
